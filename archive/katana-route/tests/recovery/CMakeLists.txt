cmake_minimum_required(VERSION 3.20)
project(RingPadRecoveryOracle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT FIXTURE_DIR OR NOT FIXTURE)
    message(FATAL_ERROR "FIXTURE_DIR and FIXTURE are required")
endif()
set(DCR_HOST_AVX2 OFF CACHE BOOL "Portable ARM64 experiment" FORCE)
add_subdirectory("${FIXTURE_DIR}" generated)
if(NOT ORACLE_SOURCE)
    set(ORACLE_SOURCE execution.cpp)
endif()
add_executable(recovery_oracle "${ORACLE_SOURCE}")
target_compile_definitions(recovery_oracle PRIVATE RECOVERY_FIXTURE=${FIXTURE})
target_link_libraries(recovery_oracle PRIVATE dreamcast_generated)

if(MODULE_JOIN_DIR)
    target_sources(recovery_oracle PRIVATE board-join.cpp)
    target_include_directories(recovery_oracle PRIVATE "${MODULE_JOIN_DIR}")
    target_compile_definitions(recovery_oracle PRIVATE RECOVERY_SBOARD_JOIN=1)
    include("${MODULE_JOIN_DIR}/module-sources.cmake")
endif()
